function Meta({ title, description }: { title: string; description: string }) {
  useEffect(() => {
    document.title = title;
    let tag = document.querySelector('meta[name="description"]');
    if (!tag) {
      tag = document.createElement("meta");
      tag.setAttribute("name", "description");
      document.head.appendChild(tag);
    }
    tag.setAttribute("content", description);
  }, [title, description]);
  return null;
}

function Catalog() {
  const queryCategory = new URLSearchParams(window.location.search).get("categoria");
  const initialCategory = categories.some((category) => category.name === queryCategory) ? queryCategory as Category : "Todos";
  const [selected, setSelected] = useState<"Todos" | Category>(initialCategory);
  const [menuOpen, setMenuOpen] = useState(false);
  const visible = useMemo(() => selected === "Todos" ? products : products.filter((product) => product.category === selected), [selected]);
  return (
    <div className="site-shell noise">
      <Meta title="Catálogo — G72 Plastic" description="Explore o catálogo de personalizados da G72 Plastic: copos, taças, canecas, chaveiros, presentes e peças 3D." />
      <Header />
      <main className="catalog-page container">
        <div className="catalog-top animate-rise">
          <div><p className="eyebrow">Peças, ideias e possibilidades</p><h1>Catálogo<br /><em>G72 Plastic.</em></h1></div>
          <p>Uma seleção do que já saiu do nosso ateliê. Gostou de uma peça? Ela pode mudar de cor, nome, tamanho e até de propósito.</p>
        </div>
        <div className="filter-bar">
          <div className="filter-scroll">
            {(["Todos", ...categories.map((category) => category.name)] as const).map((category) => (
              <button key={category} onClick={() => setSelected(category)} className={selected === category ? "selected" : ""} data-testid={`button-filter-${category === "Todos" ? "all" : category}`}>
                {category === "Todos" ? "Tudo" : category}
              </button>
            ))}
          </div>
          <button className="mobile-filter-button" onClick={() => setMenuOpen(!menuOpen)} data-testid="button-mobile-filters">{selected === "Todos" ? "Filtrar" : selected}<ChevronDown size={16} /></button>
        </div>
        {menuOpen && <div className="mobile-filter-menu">{(["Todos", ...categories.map((category) => category.name)] as const).map((category) => <button key={category} onClick={() => { setSelected(category); setMenuOpen(false); }} data-testid={`button-mobile-filter-${category}`}>{category}<Check size={16} className={selected === category ? "visible" : ""} /></button>)}</div>}
        <div className="catalog-result-line"><span>{visible.length} {visible.length === 1 ? "peça" : "peças"} encontradas</span><span className="result-dot" /><span>produzidas em pequenos lotes</span></div>

function Router() {
  return <ErrorBoundary resetKey={window.location.pathname}><Switch><Route path="/" component={Home} /><Route path="/catalogo" component={Catalog} /><Route path="/produto/:slug" component={ProductDetail} /><Route component={NotFound} /></Switch></ErrorBoundary>;
}

function App() {
  return <QueryClientProvider client={new QueryClient()}><TooltipProvider><WouterRouter base={import.meta.env.BASE_URL.replace(/\/$/, "")}><Router /></WouterRouter><Toaster /></TooltipProvider></QueryClientProvider>;
}
